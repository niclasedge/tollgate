package registry

// tollgate overlay (ADR-0020, #137, #3): the pinned embedded catalog lacks some
// current models, and -local-model plus Anthropic-only egress keep the remote
// catalog out. This file sorts after model_updater.go, so its init runs after the
// embedded catalog is loaded. Each entry clones a same-generation embedded model
// under the new id and does nothing once upstream embeds the model.

type tollgateOverlay struct {
	id, base, name string
	// adaptive replaces the base thinking with adaptive-only thinking (no
	// budget range, no zero), as upstream catalogs it for the new model.
	adaptive bool
}

// Order is fixed so the catalog stays byte-for-byte deterministic.
var tollgateOverlays = []tollgateOverlay{
	// Sonnet 5.5: 1M context, 128K output, adaptive thinking low..max.
	{id: "claude-haiku-5-5", base: "claude-sonnet-5-5", name: "Claude Haiku 5.5"},
	// Upstream v8.0.63 lists Fable 5.2 like Fable 5.1, but adaptive-only.
	{id: "claude-fable-5-2", base: "claude-fable-5-1", name: "Claude Fable 5.2", adaptive: true},
}

func init() { tollgateAddOverlays() }

func tollgateAddOverlays() {
	modelsCatalogStore.mu.Lock()
	defer modelsCatalogStore.mu.Unlock()
	data := modelsCatalogStore.data
	if data == nil {
		return
	}
	for _, o := range tollgateOverlays {
		var base *ModelInfo
		known := false
		for _, m := range data.Claude {
			switch m.ID {
			case o.id:
				known = true
			case o.base:
				base = m
			}
		}
		if known || base == nil {
			continue
		}
		m := cloneModelInfo(base)
		m.ID = o.id
		m.DisplayName = o.name
		if o.adaptive && m.Thinking != nil {
			m.Thinking = &ThinkingSupport{
				DynamicAllowed: true,
				Levels:         append([]string(nil), m.Thinking.Levels...),
			}
		}
		data.Claude = append(data.Claude, m)
	}
}
