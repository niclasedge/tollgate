package registry

import "testing"

// TestTollgateOverlay checks the ADR-0020 overlay against the pinned catalog.
func TestTollgateOverlay(t *testing.T) {
	byID := func() (map[string]*ModelInfo, map[string]int) {
		models, count := map[string]*ModelInfo{}, map[string]int{}
		for _, m := range GetClaudeModels() {
			models[m.ID] = m
			count[m.ID]++
		}
		return models, count
	}
	models, count := byID()
	for _, o := range tollgateOverlays {
		m, base := models[o.id], models[o.base]
		if m == nil || count[o.id] != 1 {
			t.Fatalf("%s listed %d times, want once", o.id, count[o.id])
		}
		if base == nil {
			t.Fatalf("base %s missing", o.base)
		}
		if m.DisplayName != o.name || m.Type != "claude" || m.ContextLength != 1000000 || m.MaxCompletionTokens != 128000 {
			t.Fatalf("unexpected entry: %+v", m)
		}
		if m.Thinking == nil || len(m.Thinking.Levels) != 5 || !m.Thinking.DynamicAllowed {
			t.Fatalf("%s: unexpected thinking: %+v", o.id, m.Thinking)
		}
		if m.Thinking == base.Thinking {
			t.Fatalf("%s shares thinking with its base", o.id)
		}
		if got := LookupStaticModelInfo(o.id); got == nil {
			t.Fatalf("static lookup misses %s", o.id)
		}
	}
	// Fable 5.2 is adaptive-only, as in upstream v8.0.63.
	if th := models["claude-fable-5-2"].Thinking; th.ZeroAllowed || th.Min != 0 || th.Max != 0 {
		t.Fatalf("fable 5.2 not adaptive-only: %+v", th)
	}

	// Idempotent: a second run, as after a catalog reload, adds nothing.
	tollgateAddOverlays()
	_, count = byID()
	for _, o := range tollgateOverlays {
		if count[o.id] != 1 {
			t.Fatalf("after rerun %s has %d entries, want 1", o.id, count[o.id])
		}
	}
}
