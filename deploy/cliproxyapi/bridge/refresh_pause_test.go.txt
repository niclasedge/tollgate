package auth

import (
	"context"
	"testing"
	"time"
)

type tollgateBlockingRefresh struct {
	mockOAuthErrorExecutor
	entered, release chan struct{}
}

func (e *tollgateBlockingRefresh) Refresh(ctx context.Context, a *Auth) (*Auth, error) {
	close(e.entered)
	select {
	case <-ctx.Done():
		return nil, ctx.Err()
	case <-e.release:
	}
	e.refreshCalls.Add(1)
	a.Metadata["refresh_token"] = "rotated-token"
	return a, nil
}
func TestTollgatePauseWaitsForRefreshAndBlocksFurtherRotation(t *testing.T) {
	ctx := context.Background()
	m := NewManager(nil, &RoundRobinSelector{}, nil)
	e := &tollgateBlockingRefresh{mockOAuthErrorExecutor: mockOAuthErrorExecutor{id: "test-provider"}, entered: make(chan struct{}), release: make(chan struct{})}
	m.RegisterExecutor(e)
	a := &Auth{ID: "pause-test", Provider: e.id, Status: StatusActive, Metadata: map[string]any{"access_token": "access", "refresh_token": "old-token"}}
	if _, err := m.Register(ctx, a); err != nil {
		t.Fatal(err)
	}
	refreshed := make(chan error, 1)
	go func() { _, err := m.ForceRefreshAuth(ctx, a.ID); refreshed <- err }()
	<-e.entered
	paused := make(chan error, 1)
	go func() { paused <- m.TollgatePauseRefresh(ctx, a.ID, true) }()
	select {
	case err := <-paused:
		t.Fatal("pause acknowledged before refresh completed", err)
	case <-time.After(25 * time.Millisecond):
	}
	close(e.release)
	if err := <-refreshed; err != nil {
		t.Fatal(err)
	}
	if err := <-paused; err != nil {
		t.Fatal(err)
	}
	m.mu.RLock()
	token := m.auths[a.ID].Metadata["refresh_token"]
	m.mu.RUnlock()
	if token != "rotated-token" {
		t.Fatal("latest token lost")
	}
	if _, err := m.ForceRefreshAuth(ctx, a.ID); err == nil {
		t.Fatal("refresh entered paused executor")
	}
	if e.refreshCalls.Load() != 1 {
		t.Fatal("rotation while paused")
	}
	if err := m.TollgatePauseRefresh(ctx, a.ID, false); err != nil {
		t.Fatal(err)
	}
	if m.tollgateRefreshPaused(a.ID) {
		t.Fatal("resume failed")
	}
}

func TestTollgatePauseFailsClosedOnPersistenceFailure(t *testing.T) {
	m := NewManager(&persistFailureStore{}, &RoundRobinSelector{}, nil)
	a := &Auth{ID: "persist-pause", Provider: "claude", Status: StatusActive, Metadata: map[string]any{"access_token": "access", "refresh_token": "refresh"}}
	if _, err := m.Register(context.Background(), a); err != nil {
		t.Fatal(err)
	}
	if err := m.TollgatePauseRefresh(context.Background(), a.ID, true); err == nil {
		t.Fatal("pause acknowledged without durable snapshot")
	}
}
func TestTollgatePauseCanReserveFutureCredential(t *testing.T) {
	m := NewManager(nil, &RoundRobinSelector{}, nil)
	ctx := context.Background()
	if err := m.TollgatePauseRefresh(ctx, "future-file.json", true); err != nil {
		t.Fatal(err)
	}
	e := &mockOAuthErrorExecutor{id: "test-provider"}
	m.RegisterExecutor(e)
	if _, err := m.Register(ctx, &Auth{ID: "future-file.json", Provider: e.id, Status: StatusActive, Metadata: map[string]any{"access_token": "access", "refresh_token": "refresh"}}); err != nil {
		t.Fatal(err)
	}
	if _, err := m.ForceRefreshAuth(ctx, "future-file.json"); err == nil || e.refreshCalls.Load() != 0 {
		t.Fatal("new upload refreshed before verification")
	}
	if err := m.TollgatePauseRefresh(ctx, "future-file.json", false); err != nil {
		t.Fatal(err)
	}
	if _, err := m.ForceRefreshAuth(ctx, "future-file.json"); err != nil || e.refreshCalls.Load() != 1 {
		t.Fatal("resume did not release refresh", err)
	}
}
